# Radius

export const radii = [
  { name: 'rounded-sm', v: 'var(--tomo-radius-sm)', px: '4px' },
  { name: 'rounded-md', v: 'var(--tomo-radius-md)', px: '8px' },
  { name: 'rounded-full', v: 'var(--tomo-radius-full)', px: '9999px' },
];

<div className="flex gap-32 font-sans">
  {radii.map(({ name, v, px }) => (
    <div key={name} className="flex flex-col items-center gap-8">
      <div
        className="size-[80px] border border-border-strong bg-bg-subtle"
        style={{ borderRadius: v }}
      />
      <code>{name}</code>
      <span className="text-caption text-text-subtle">{px}</span>
    </div>
  ))}
</div>
